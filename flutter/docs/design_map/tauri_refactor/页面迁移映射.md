## J1: 认证旅程

### BootstrapPage（引导页面）

```tsx
// src/pages/auth/BootstrapPage.tsx
import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { useAuthStore } from '@/stores/authStore';
import { GlassCard } from '@/components/liquid-glass/GlassCard';
import { Button } from '@/components/ui/Button';
import { Input } from '@/components/ui/Input';
import { PasswordInput } from '@/components/forms/PasswordInput';

export function BootstrapPage() {
  const navigate = useNavigate();
  const { bootstrap, isLoading, error } = useAuthStore();
  const [accountName, setAccountName] = useState('');
  const [password, setPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState('');

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (password !== confirmPassword) return;
    await bootstrap(accountName, password);
    navigate('/');
  };

  return (
    <div className="bootstrap-page">
      <GlassCard className="bootstrap-card">
        <h1>欢迎使用 SoloSoul</h1>
        <p>首次使用，请创建主密码保护您的数据</p>
        
        <form onSubmit={handleSubmit}>
          <Input
            label="账户名称"
            value={accountName}
            onChange={setAccountName}
            placeholder="例如：个人账户"
          />
          <PasswordInput
            label="主密码"
            value={password}
            onChange={setPassword}
            hint="至少 8 位，包含大小写字母和数字"
          />
          <PasswordInput
            label="确认密码"
            value={confirmPassword}
            onChange={setConfirmPassword}
          />
          
          {error && <div className="error">{error}</div>}
          
          <Button type="submit" loading={isLoading}>
            创建账户
          </Button>
        </form>
      </GlassCard>
    </div>
  );
}
```

### LoginPage（登录页面）

```tsx
// src/pages/auth/LoginPage.tsx
import { useState, useEffect } from 'react';
import { useNavigate } from 'react-router-dom';
import { useAuthStore } from '@/stores/authStore';
import { useVaultStore } from '@/stores/vaultStore';
import { GlassCard } from '@/components/liquid-glass/GlassCard';

export function LoginPage() {
  const navigate = useNavigate();
  const { login, checkHasAccount, isAuthenticated } = useAuthStore();
  const { accounts, loadAccounts } = useVaultStore();
  const [selectedAccount, setSelectedAccount] = useState('');
  const [password, setPassword] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  useEffect(() => {
    // 检查是否已有账户
    checkHasAccount().then((hasAccount) => {
      if (!hasAccount) {
        navigate('/bootstrap');
      } else {
        loadAccounts();
      }
    });
  }, []);

  useEffect(() => {
    if (isAuthenticated) {
      navigate('/');
    }
  }, [isAuthenticated]);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setIsLoading(true);
    await login(selectedAccount, password);
    setIsLoading(false);
  };

  return (
    <div className="login-page">
      <GlassCard className="login-card">
        <h1>🔐 解锁 Vault</h1>
        
        <form onSubmit={handleSubmit}>
          <select
            value={selectedAccount}
            onChange={(e) => setSelectedAccount(e.target.value)}
          >
            <option value="">选择账户</option>
            {accounts.map((acc) => (
              <option key={acc.id} value={acc.id}>
                {acc.name}
              </option>
            ))}
          </select>
          
          <input
            type="password"
            value={password}
            onChange={(e) => setPassword(e.target.value)}
            placeholder="输入主密码"
          />
          
          <button type="submit" disabled={isLoading}>
            {isLoading ? '解锁中...' : '解锁'}
          </button>
        </form>
      </GlassCard>
    </div>
  );
}
```

### ObjectWorkspacePage（对象工作区）

```tsx
// src/pages/workspace/ObjectWorkspacePage.tsx
import { useState, useEffect } from 'react';
import { useSearchParams, useNavigate } from 'react-router-dom';
import { useUnifiedObjectStore } from '@/stores/unifiedObjectStore';
import { AppShell } from '@/components/layout/AppShell';
import { GlassCard } from '@/components/liquid-glass/GlassCard';
import { ObjectListItem } from '@/components/data-display/ObjectListItem';
import { CategoryFilter } from '@/components/ui/CategoryFilter';

export function ObjectWorkspacePage() {
  const navigate = useNavigate();
  const [searchParams] = useSearchParams();
  const sectionFilter = searchParams.get('section');
  
  const { objects, loadObjects, isLoading } = useUnifiedObjectStore();
  const [searchQuery, setSearchQuery] = useState('');

  useEffect(() => {
    loadObjects({ sectionType: sectionFilter || undefined });
  }, [sectionFilter]);

  const filteredObjects = objects.filter((obj) =>
    obj.name.toLowerCase().includes(searchQuery.toLowerCase())
  );

  return (
    <AppShell
      title="对象工作区"
      actions={
        <button onClick={() => navigate('/editor')}>
          ➕ 新建
        </button>
      }
    >
      <CategoryFilter
        selected={sectionFilter}
        onSelect={(cat) => navigate(`/workspace?section=${cat}`)}
      />
      
      <input
        type="search"
        placeholder="搜索对象..."
        value={searchQuery}
        onChange={(e) => setSearchQuery(e.target.value)}
      />

      <div className="object-list">
        {isLoading ? (
          <div>加载中...</div>
        ) : filteredObjects.length === 0 ? (
          <GlassCard className="empty-state">
            <p>暂无对象</p>
            <button onClick={() => navigate('/editor')}>
              创建第一个对象
            </button>
          </GlassCard>
        ) : (
          filteredObjects.map((obj) => (
            <ObjectListItem
              key={obj.id}
              object={obj}
              onClick={() => navigate(`/editor/${obj.id}`)}
            />
          ))
        )}
      </div>
    </AppShell>
  );
}
```

### ObjectEditorPage（对象编辑器）

```tsx
// src/pages/editor/ObjectEditorPage.tsx
import { useState, useEffect, useCallback } from 'react';
import { useParams, useNavigate } from 'react-router-dom';
import { useUnifiedObjectStore } from '@/stores/unifiedObjectStore';
import { useProfileStore } from '@/stores/profileStore';
import { AppShell } from '@/components/layout/AppShell';
import { GlassCard } from '@/components/liquid-glass/GlassCard';
import { SectionEditor } from './section-panels/SectionEditor';
import { FieldEditor } from '@/components/forms/FieldEditor';
import { useDebounce } from '@/hooks/useDebounce';

export function ObjectEditorPage() {
  const { objectId } = useParams<{ objectId: string }>();
  const navigate = useNavigate();
  const isNew = !objectId;

  const { getObject, createObject, updateObject } = useUnifiedObjectStore();
  const [object, setObject] = useState<UnifiedObject | null>(null);
  const [isLoading, setIsLoading] = useState(!isNew);

  // 防抖保存
  const debouncedSave = useDebounce(
    useCallback(async (data: Partial<UnifiedObject>) => {
      if (isNew) return;
      await updateObject(objectId!, data);
    }, [isNew, objectId]),
    500
  );

  useEffect(() => {
    if (!isNew && objectId) {
      getObject(objectId).then((obj) => {
        setObject(obj);
        setIsLoading(false);
      });
    }
  }, [objectId]);

  const handleFieldChange = (sectionType: string, fieldKey: string, value: unknown) => {
    setObject((prev) => {
      if (!prev) return prev;
      const updated = {
        ...prev,
        sectionData: {
          ...prev.sectionData,
          [sectionType]: {
            ...prev.sectionData[sectionType],
            [fieldKey]: value,
          },
        },
      };
      debouncedSave(updated);
      return updated;
    });
  };

  if (isLoading) return <div>加载中...</div>;

  return (
    <AppShell title={isNew ? '新建对象' : object?.name || '编辑对象'}>
      <div className="editor-layout">
        {/* 对象元数据 */}
        <GlassCard>
          <input
            value={object?.name || ''}
            onChange={(e) => handleFieldChange('meta', 'name', e.target.value)}
            placeholder="对象名称"
          />
          <select
            value={object?.sectionType || ''}
            onChange={(e) => handleFieldChange('meta', 'sectionType', e.target.value)}
          >
            <option value="">选择分区类型</option>
            <option value="passport">护照</option>
            <option value="identity">身份</option>
            {/* ... */}
          </select>
        </GlassCard>

        {/* 分区编辑器 */}
        {object?.sectionType && (
          <SectionEditor
            sectionType={object.sectionType}
            data={object.sectionData[object.sectionType] || {}}
            onChange={(key, value) => handleFieldChange(object.sectionType!, key, value)}
          />
        )}

        {/* 操作按钮 */}
        <div className="editor-actions">
          {isNew ? (
            <button onClick={() => createObject(object!)}>创建</button>
          ) : (
            <button onClick={() => updateObject(objectId!, object!)}>
              保存
            </button>
          )}
          <button onClick={() => navigate(-1)}>取消</button>
        </div>
      </div>
    </AppShell>
  );
}
```

### PasswordVerificationDialog（密码验证对话框）

```tsx
// src/components/dialogs/PasswordVerificationDialog.tsx
import { useState } from 'react';
import { Dialog } from '@/components/ui/Dialog';
import { PasswordInput } from '@/components/forms/PasswordInput';
import { useVaultStore } from '@/stores/vaultStore';

interface PasswordVerificationDialogProps {
  isOpen: boolean;
  onClose: () => void;
  onVerified: () => void;
  title?: string;
}

export function PasswordVerificationDialog({
  isOpen,
  onClose,
  onVerified,
  title = '验证身份',
}: PasswordVerificationDialogProps) {
  const { unlock } = useVaultStore();
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const [isLoading, setIsLoading] = useState(false);

  const handleVerify = async () => {
    setIsLoading(true);
    setError('');
    try {
      // 使用当前账户解锁
      const accountId = useAuthStore.getState().currentAccount?.id;
      if (!accountId) throw new Error('未登录');
      await unlock(accountId, password);
      onVerified();
      onClose();
    } catch {
      setError('密码不正确');
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <Dialog isOpen={isOpen} onClose={onClose}>
      <h2>{title}</h2>
      <p>此操作需要验证您的身份</p>
      <PasswordInput
        value={password}
        onChange={setPassword}
        placeholder="输入主密码"
      />
      {error && <div className="error">{error}</div>}
      <div className="dialog-actions">
        <button onClick={onClose}>取消</button>
        <button onClick={handleVerify} disabled={isLoading}>
          {isLoading ? '验证中...' : '验证'}
        </button>
      </div>
    </Dialog>
  );
}
```

### SensitiveValue（敏感数据遮罩）

```tsx
// src/components/data-display/SensitiveValue.tsx
import { useState } from 'react';
import { useSettingsStore } from '@/stores/settingsStore';

interface SensitiveValueProps {
  value: string;
  level?: 'public' | 'internal' | 'sensitive' | 'critical';
  maskChar?: string;
}

export function SensitiveValue({
  value,
  level = 'internal',
  maskChar = '•',
}: SensitiveValueProps) {
  const [isVisible, setIsVisible] = useState(false);
  const { settings } = useSettingsStore();

  // 敏感度 ≤ internal 不遮罩
  if (level === 'internal') {
    return <span>{value}</span>;
  }

  const displayValue = isVisible ? value : maskChar.repeat(Math.min(value.length, 8));

  return (
    <span className="sensitive-value">
      {displayValue}
      <button
        className="toggle-visibility"
        onClick={() => setIsVisible(!isVisible)}
      >
        {isVisible ? '🙈' : '👁️'}
      </button>
    </span>
  );
}
```

## 迁移优先级建议

### P0（必须先实现）

1. **BootstrapPage** — 应用入口，无依赖
2. **LoginPage** — 认证核心，依赖 Vault IPC
3. **HomePage** — 首页，依赖 Profile 数据
4. **ObjectWorkspacePage** — 核心功能，依赖 UnifiedObject IPC
5. **ObjectEditorPage** — 核心功能，最复杂页面

### P1（重要功能）

6. **SettingsPage** — 设置导航
7. **SecuritySettingsPage** — 安全功能
8. **SensitivitySettingsPage** — 隐私功能
9. **SearchPage** — 全局搜索

### P2（辅助功能）

10. **DataManagementPage** — 数据管理
11. **ExportImportPage** — 导入导出
12. **BackupConfigPage** — 备份
