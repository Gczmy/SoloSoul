```rust
// src-tauri/src/db/connection.rs
use rusqlite::{Connection, Result};
use std::path::Path;
use std::sync::{Arc, Mutex};

pub struct DbPool {
    connections: Arc<Mutex<Vec<Connection>>>,
    db_path: String,
}

impl DbPool {
    pub fn new(db_path: &Path) -> Result<Self> {
        Ok(Self {
            connections: Arc::new(Mutex::new(Vec::new())),
            db_path: db_path.to_string_lossy().to_string(),
        })
    }

    pub fn get_connection(&self) -> Result<Connection> {
        let mut conns = self.connections.lock().unwrap();
        if let Some(conn) = conns.pop() {
            Ok(conn)
        } else {
            Connection::open(&self.db_path)
        }
    }

    pub fn return_connection(&self, conn: Connection) {
        let mut conns = self.connections.lock().unwrap();
        conns.push(conn);
    }
}

// 线程安全的连接获取
pub struct PooledConnection {
    conn: Option<Connection>,
    pool: Arc<DbPool>,
}

impl Drop for PooledConnection {
    fn drop(&mut self) {
        if let Some(conn) = self.conn.take() {
            self.pool.return_connection(conn);
        }
    }
}

impl std::ops::Deref for PooledConnection {
    type Target = Connection;
    fn deref(&self) -> &Self::Target {
        self.conn.as_ref().unwrap()
    }
}

impl std::ops::DerefMut for PooledConnection {
    fn deref_mut(&mut self) -> &mut Self::Target {
        self.conn.as_mut().unwrap()
    }
}
```

```rust
use rusqlite::{Connection, Result};

pub const CURRENT_SCHEMA_VERSION: i32 = 6;

pub fn run_migrations(conn: &Connection) -> Result<()> {
    let user_version: i32 = conn.query_row("PRAGMA user_version", [], |row| row.get(0))?;
    
    if user_version < 1 {
        migrate_v0_to_v1(conn)?;
    }
    if user_version < 2 {
        migrate_v1_to_v2(conn)?;
    }
    if user_version < 6 {
        migrate_v5_to_v6(conn)?;
    }
    
    conn.execute(&format!("PRAGMA user_version = {}", CURRENT_SCHEMA_VERSION), [])?;
    Ok(())
}
```

```rust
use crate::db::repositories::profile_repo::ProfileRepository;
use crate::core::profile::model::{ProfileData, SectionData, FieldValue};
use crate::core::utils::error::AppError;
use rusqlite::Connection;

pub struct ProfileService<'a> {
    conn: &'a Connection,
}

impl<'a> ProfileService<'a> {
    pub fn new(conn: &'a Connection) -> Self {
        Self { conn }
    }

    pub fn get_profile(&self, account_id: &str) -> Result<ProfileData, AppError> {
        let repo = ProfileRepository::new(self.conn);
        match repo.get_by_account(account_id)? {
            Some(profile) => Ok(profile),
            None => Err(AppError::Other("Profile not found".to_string())),
        }
    }

    pub fn update_profile(&self, data: &ProfileData) -> Result<(), AppError> {
        let repo = ProfileRepository::new(self.conn);
        repo.upsert(data)?;
        Ok(())
    }

    pub fn update_field(
        &self,
        account_id: &str,
        section_type: &str,
        field_key: &str,
        value: &FieldValue,
    ) -> Result<(), AppError> {
        let repo = ProfileRepository::new(self.conn);
        
        let mut profile = match repo.get_by_account(account_id)? {
            Some(p) => p,
            None => return Err(AppError::Other("Profile not found".to_string())),
        };

        // 更新 JSON 字段
        profile.update_json_field(section_type, field_key, value)?;
        profile.updated_at = Some(chrono::Utc::now().to_rfc3339());

        repo.upsert(&profile)?;
        Ok(())
    }
}
```

```rust
// src-tauri/src/services/unified_object_service.rs
use crate::db::repositories::unified_object_repo::UnifiedObjectRepository;
use crate::core::profile::model::{UnifiedObject, ObjectFilter, SortConfig};
use crate::core::utils::error::AppError;
use rusqlite::Connection;

pub struct UnifiedObjectService<'a> {
    conn: &'a Connection,
}

impl<'a> UnifiedObjectService<'a> {
    pub fn new(conn: &'a Connection) -> Self {
        Self { conn }
    }

    pub fn list(
        &self,
        account_id: &str,
        filter: Option<ObjectFilter>,
        sort: Option<SortConfig>,
    ) -> Result<Vec<UnifiedObject>, AppError> {
        let repo = UnifiedObjectRepository::new(self.conn);
        repo.list(account_id, filter, sort)
            .map_err(|e| AppError::Database(e))
    }

    pub fn get(&self, account_id: &str, object_id: &str) -> Result<UnifiedObject, AppError> {
        let repo = UnifiedObjectRepository::new(self.conn);
        match repo.get_by_id(account_id, object_id)? {
            Some(obj) => Ok(obj),
            None => Err(AppError::Other("Object not found".to_string())),
        }
    }

    pub fn create(
        &self,
        account_id: &str,
        data: UnifiedObjectInput,
    ) -> Result<UnifiedObject, AppError> {
        let repo = UnifiedObjectRepository::new(self.conn);
        let object = UnifiedObject {
            id: generate_uuid(),
            account_id: account_id.to_string(),
            name: data.name,
            section_type: data.section_type,
            section_data: data.section_data,
            sensitivity_level: data.sensitivity_level.unwrap_or("internal".to_string()),
            created_at: Some(chrono::Utc::now().to_rfc3339()),
            updated_at: Some(chrono::Utc::now().to_rfc3339()),
            deleted_at: None,
        };
        repo.insert(&object)?;
        Ok(object)
    }

    pub fn update(
        &self,
        account_id: &str,
        object_id: &str,
        data: UnifiedObjectInput,
    ) -> Result<UnifiedObject, AppError> {
        let repo = UnifiedObjectRepository::new(self.conn);
        let mut object = self.get(account_id, object_id)?;
        
        object.name = data.name;
        object.section_type = data.section_type;
        object.section_data = data.section_data;
        if let Some(level) = data.sensitivity_level {
            object.sensitivity_level = level;
        }
        object.updated_at = Some(chrono::Utc::now().to_rfc3339());
        
        repo.update(&object)?;
        Ok(object)
    }

    pub fn delete(&self, account_id: &str, object_id: &str) -> Result<(), AppError> {
        let repo = UnifiedObjectRepository::new(self.conn);
        repo.soft_delete(account_id, object_id)?;
        Ok(())
    }

    pub fn permanently_delete(&self, account_id: &str, object_id: &str) -> Result<(), AppError> {
        let repo = UnifiedObjectRepository::new(self.conn);
        repo.hard_delete(account_id, object_id)?;
        Ok(())
    }
}
```

```rust
// src-tauri/src/services/search_service.rs
use crate::db::repositories::{profile_repo::ProfileRepository, unified_object_repo::UnifiedObjectRepository};
use crate::core::profile::model::{SearchResult, SearchResultItem, SearchOptions};
use crate::core::utils::error::AppError;
use rusqlite::Connection;

pub struct SearchService<'a> {
    conn: &'a Connection,
}

impl<'a> SearchService<'a> {
    pub fn new(conn: &'a Connection) -> Self {
        Self { conn }
    }

    pub fn unified_search(
        &self,
        account_id: &str,
        query: &str,
        options: SearchOptions,
    ) -> Result<SearchResult, AppError> {
        let mut items = Vec::new();
        
        // 1. 搜索 Profile 字段
        let profile_repo = ProfileRepository::new(self.conn);
        if let Some(profile) = profile_repo.get_by_account(account_id)? {
            items.extend(self.search_profile_fields(&profile, query));
        }
        
        // 2. 搜索 UnifiedObjects
        let object_repo = UnifiedObjectRepository::new(self.conn);
        let objects = object_repo.search(account_id, query, &options)?;
        for obj in objects {
            items.push(SearchResultItem {
                object_id: obj.id,
                name: obj.name,
                section_type: obj.section_type,
                matched_field: None,
                matched_value: Some(query.to_string()),
                relevance: 1.0,
            });
        }
        
        // 3. 按相关性排序
        items.sort_by(|a, b| b.relevance.partial_cmp(&a.relevance).unwrap());
        
        let total = items.len();
        let limit = options.limit.unwrap_or(50);
        let has_more = total > limit;
        items.truncate(limit);
        
        Ok(SearchResult { items, total, has_more })
    }

    fn search_profile_fields(&self, profile: &ProfileData, query: &str) -> Vec<SearchResultItem> {
        let mut items = Vec::new();
        let query_lower = query.to_lowercase();
        
        // 搜索基本字段
        let fields = [
            ("full_name", profile.full_name.as_deref()),
            ("email", profile.email.as_deref()),
            ("phone", profile.phone.as_deref()),
            ("identity_number", profile.identity_number.as_deref()),
        ];
        
        for (field_name, value) in fields {
            if let Some(v) = value {
                if v.to_lowercase().contains(&query_lower) {
                    items.push(SearchResultItem {
                        object_id: profile.account_id.clone(),
                        name: v.to_string(),
                        section_type: "profile".to_string(),
                        matched_field: Some(field_name.to_string()),
                        matched_value: Some(v.to_string()),
                        relevance: 1.0,
                    });
                }
            }
        }
        
        items
    }
}
```

```rust
// src-tauri/src/services/user_data_service.rs
use crate::core::vault::Vault;
use crate::core::user_data::{UserPreferences, AuditLog, SearchHistory};
use crate::core::utils::error::AppError;
use serde_json::Value;

/// 用户数据服务 —— 统一处理偏好数据和行为数据的加密读写
/// 
/// 所有数据通过 Vault 主密钥 AES-256-GCM 加密，Vault 锁定后不可访问。
pub struct UserDataService<'a> {
    vault: &'a Vault,
}

impl<'a> UserDataService<'a> {
    pub fn new(vault: &'a Vault) -> Self {
        Self { vault }
    }

    // ==================== 偏好数据（Preferences） ====================

    pub fn get_preferences(&self) -> Result<UserPreferences, AppError> {
        self.vault.load_preferences()
            .map_err(|e| AppError::Vault(e.to_string()))
    }

    pub fn get_preference(&self, key: &str) -> Result<Option<Value>, AppError> {
        let prefs = self.get_preferences()?;
        let value = match key {
            "theme" => serde_json::to_value(prefs.theme).ok(),
            "locale" => serde_json::to_value(prefs.locale).ok(),
            "autoLockTimeoutMinutes" => serde_json::to_value(prefs.auto_lock_timeout_minutes).ok(),
            "biometricEnabled" => serde_json::to_value(prefs.biometric_enabled).ok(),
            "confirmDelete" => serde_json::to_value(prefs.confirm_delete).ok(),
            "hapticFeedback" => serde_json::to_value(prefs.haptic_feedback).ok(),
            "developerMode" => serde_json::to_value(prefs.developer_mode).ok(),

            "fontSize" => serde_json::to_value(prefs.font_size).ok(),

            "timezone" => serde_json::to_value(prefs.timezone).ok(),
            "defaultSensitivityStrategy" => serde_json::to_value(prefs.default_sensitivity_strategy).ok(),
            _ => None,
        };
        Ok(value)
    }

    pub fn update_preference(
        &self,
        key: &str,
        value: Value,
    ) -> Result<(), AppError> {
        let mut prefs = self.get_preferences()?;
        
        match key {
            "theme" => prefs.theme = serde_json::from_value(value)?,
            "locale" => prefs.locale = serde_json::from_value(value)?,
            "autoLockTimeoutMinutes" => {
                prefs.auto_lock_timeout_minutes = serde_json::from_value(value)?
            }
            "biometricEnabled" => prefs.biometric_enabled = serde_json::from_value(value)?,
            "confirmDelete" => prefs.confirm_delete = serde_json::from_value(value)?,
            "hapticFeedback" => prefs.haptic_feedback = serde_json::from_value(value)?,
            "developerMode" => prefs.developer_mode = serde_json::from_value(value)?,
            "accentColor" => prefs.accent_color = serde_json::from_value(value)?,
            "fontSize" => prefs.font_size = serde_json::from_value(value)?,
            "density" => prefs.density = serde_json::from_value(value)?,
            "dateFormat" => prefs.date_format = serde_json::from_value(value)?,
            "currencyFormat" => prefs.currency_format = serde_json::from_value(value)?,
            "timezone" => prefs.timezone = serde_json::from_value(value)?,
            "defaultSensitivityStrategy" => {
                prefs.default_sensitivity_strategy = serde_json::from_value(value)?
            }
            _ => return Err(AppError::InvalidParameter(format!("Unknown preference key: {}", key))),
        }
        
        prefs.updated_at = chrono::Utc::now();
        
        self.vault.save_preferences(&prefs)
            .map_err(|e| AppError::Vault(e.to_string()))?;
        
        // 写入审计日志：偏好变更
        self.vault.append_audit_log(AuditLogEntry {
            id: generate_uuid(),
            timestamp: chrono::Utc::now(),
            action: AuditAction::PreferenceUpdated,
            entity_type: Some("user_preference".to_string()),
            entity_id: Some(key.to_string()),
            description: format!("偏好 '{}' 已更新", key),
            performed_by: "user".to_string(),
            sensitivity_level: SensitivityLevel::Sensitive,
        }).map_err(|e| AppError::Vault(e.to_string()))?;
        
        Ok(())
    }

    pub fn reset_preferences_to_default(&self) -> Result<(), AppError> {
        let prefs = UserPreferences::default();
```
