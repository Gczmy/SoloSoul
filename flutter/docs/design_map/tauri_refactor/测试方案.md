```rust
// crates/solosoul-crypto/src/kdf_test.rs
#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_kdf_known_vector() {
        // 已知测试向量
        let password = "password";
        let salt = hex::decode("000102030405060708090a0b0c0d0e0f").unwrap();
        let config = KdfConfig {
            memory_kb: 32,
            iterations: 3,
            parallelism: 4,
        };

        let key = derive_key(password, &salt, &config).unwrap();
        assert_eq!(key.len(), 32);
        
        // 与参考实现对比
        let expected = hex::decode("...").unwrap();
        assert_eq!(key.as_slice(), expected.as_slice());
    }

    #[test]
    fn test_kdf_different_passwords() {
        let salt = generate_salt();
        let config = KdfConfig::development();

        let key1 = derive_key("password1", &salt, &config).unwrap();
        let key2 = derive_key("password2", &salt, &config).unwrap();

        assert_ne!(key1.as_slice(), key2.as_slice());
    }
}
```

### Vault 测试（使用临时目录）

```rust
// src-tauri/src/已废弃/vault_test.rs
#[cfg(test)]
mod tests {
    use super::*;
    use tempfile::tempdir;

    #[tokio::test]
    async fn test_vault_lifecycle() {
        let temp = tempdir().unwrap();
        let vault = VaultManager::new(temp.path()).unwrap();

        // Initialize
        vault.initialize("test", "password").await.unwrap();
        assert_eq!(vault.state().await, VaultState::Unlocked);

        // Lock
        vault.lock().await.unwrap();
        assert_eq!(vault.state().await, VaultState::Locked);

        // Unlock
        vault.unlock("test", "password").await.unwrap();
        assert_eq!(vault.state().await, VaultState::Unlocked);

        // Change password
        vault.change_password("password", "new_password").await.unwrap();
        vault.lock().await.unwrap();
        
        // Unlock with new password
        vault.unlock("test", "new_password").await.unwrap();
        assert_eq!(vault.state().await, VaultState::Unlocked);

        // Wrong password
        let result = vault.unlock("test", "wrong").await;
        assert!(result.is_err());
    }

    #[tokio::test]
    async fn test_vault_persistence() {
        let temp = tempdir().unwrap();
        
        // First instance: initialize
        {
            let vault = VaultManager::new(temp.path()).unwrap();
            vault.initialize("test", "password").await.unwrap();
            vault.lock().await.unwrap();
        }

        // Second instance: unlock with same password
        {
            let vault = VaultManager::new(temp.path()).unwrap();
            vault.unlock("test", "password").await.unwrap();
            assert_eq!(vault.state().await, VaultState::Unlocked);
        }
    }
}
```

### Repository 测试（内存数据库）

```rust
// src-tauri/src/db/repositories/profile_repo_test.rs
#[cfg(test)]
mod tests {
    use super::*;
    use rusqlite::Connection;

    fn setup_db() -> Connection {
        let conn = Connection::open_in_memory().unwrap();
        crate::db::migrations::run_migrations(&conn).unwrap();
        conn
    }

    #[test]
    fn test_profile_crud() {
        let conn = setup_db();
        let repo = ProfileRepository::new(&conn);

        // Create
        let profile = ProfileData {
            account_id: "test_account".to_string(),
            full_name: Some("Test User".to_string()),
            ..Default::default()
        };
        repo.upsert(&profile).unwrap();

        // Read
        let retrieved = repo.get_by_account("test_account").unwrap();
        assert!(retrieved.is_some());
        assert_eq!(retrieved.unwrap().full_name, Some("Test User".to_string()));

        // Update
        let mut updated = profile.clone();
        updated.full_name = Some("Updated Name".to_string());
        repo.upsert(&updated).unwrap();

        let retrieved = repo.get_by_account("test_account").unwrap().unwrap();
        assert_eq!(retrieved.full_name, Some("Updated Name".to_string()));
    }
}
```

### Store 测试

```typescript
// src/stores/authStore.test.ts
import { useAuthStore } from './authStore';
import { describe, it, expect, vi, beforeEach } from 'vitest';
import { commands } from '@/lib/ipc';

vi.mock('@/lib/ipc');

describe('authStore', () => {
  beforeEach(() => {
    useAuthStore.setState({
      isAuthenticated: false,
      isLoading: false,
      currentAccount: null,
      error: null,
    });
  });

  it('login sets authenticated state', async () => {
    const mockAccount = { id: 'test', name: 'Test' };
    vi.mocked(commands.authLogin).mockResolvedValue(mockAccount);

    await useAuthStore.getState().login('test', 'password');

    expect(useAuthStore.getState().isAuthenticated).toBe(true);
    expect(useAuthStore.getState().currentAccount).toEqual(mockAccount);
  });

  it('login handles error', async () => {
    vi.mocked(commands.authLogin).mockRejectedValue(new Error('Invalid password'));

    await useAuthStore.getState().login('test', 'wrong');

    expect(useAuthStore.getState().isAuthenticated).toBe(false);
    expect(useAuthStore.getState().error).toBe('Invalid password');
  });

  it('logout clears state', async () => {
    vi.mocked(commands.authLogout).mockResolvedValue(undefined);
    useAuthStore.setState({ isAuthenticated: true, currentAccount: { id: 'test', name: 'Test' } });

    await useAuthStore.getState().logout();

    expect(useAuthStore.getState().isAuthenticated).toBe(false);
    expect(useAuthStore.getState().currentAccount).toBeNull();
  });
});
```

```typescript
// tests/e2e/auth.spec.ts
import { test, expect } from '@playwright/test';

test.describe('Authentication Journey', () => {
  test('first time user bootstrap', async ({ page }) => {
    await page.goto('/bootstrap');
    
    await expect(page.locator('h1')).toContainText('欢迎使用 SoloSoul');
    
    await page.fill('[name="accountName"]', 'My Account');
    await page.fill('[name="password"]', 'SecurePass123!');
    await page.fill('[name="confirmPassword"]', 'SecurePass123!');
    await page.click('button[type="submit"]');
    
    await expect(page).toHaveURL('/');
    await expect(page.locator('.welcome-card')).toContainText('欢迎回来');
  });

  test('login and logout', async ({ page }) => {
    // 先创建账户
    await page.goto('/bootstrap');
    await page.fill('[name="accountName"]', 'Test');
    await page.fill('[name="password"]', 'password');
    await page.fill('[name="confirmPassword"]', 'password');
    await page.click('button[type="submit"]');
    
    await expect(page).toHaveURL('/');
    
    // 登出
    await page.click('[data-testid="lock-button"]');
    await expect(page).toHaveURL('/login');
    
    // 重新登录
    await page.selectOption('select', 'Test');
    await page.fill('[type="password"]', 'password');
    await page.click('button[type="submit"]');
    
    await expect(page).toHaveURL('/');
  });
});
```

## CI/CD 集成

### GitHub Actions 配置

```yaml
# .github/workflows/ci.yml
name: CI

on:
  push:
    branches: [main, develop]
  pull_request:

jobs:
  rust-check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@stable
        with:
          components: rustfmt, clippy
      - uses: Swatinem/rust-cache@v2
      - name: Check formatting
        run: cargo fmt --check
      - name: Run clippy
        run: cargo clippy --workspace -- -D warnings
      - name: Run tests
        run: cargo test --workspace --verbose

  rust-coverage:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@stable
      - uses: Swatinem/rust-cache@v2
      - name: Install tarpaulin
        run: cargo install cargo-tarpaulin
      - name: Generate coverage
        run: cargo tarpaulin --workspace --out Xml --out Html
      - name: Upload coverage
        uses: codecov/codecov-action@v4
        with:
          files: ./cobertura.xml

  frontend-check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: actions/cache@v4
        with:
          path: node_modules
          key: ${{ runner.os }}-node-${{ hashFiles('package-lock.json') }}
      - name: Install dependencies
        run: npm ci
      - name: Type check
        run: npx tsc --noEmit
      - name: Lint
        run: npm run lint
      - name: Unit tests
        run: npx vitest run --coverage

  e2e-test:
    runs-on: macos-latest
    steps:
      - uses: actions/checkout@v4
      - uses: dtolnay/rust-toolchain@stable
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - uses: Swatinem/rust-cache@v2
      - uses: actions/cache@v4
        with:
          path: node_modules
          key: ${{ runner.os }}-node-${{ hashFiles('package-lock.json') }}
      - name: Install dependencies
        run: npm ci
      - name: Install Playwright
        run: npx playwright install
      - name: Build Tauri
        run: npm run tauri build -- --debug
      - name: Run E2E tests
        run: npx playwright test
      - name: Upload test results
        if: failure()
        uses: actions/upload-artifact@v4
        with:
          name: playwright-report
          path: playwright-report/
```

## 从零实现顺序

1. **配置 Rust 测试环境**
   - `cargo test` 基础配置
   - `tempfile` 用于临时目录
   - 内存数据库用于 Repository 测试

2. **编写密码学测试**
   - KDF 已知向量测试
   - AES-GCM 往返测试
   - 篡改检测测试

3. **编写 Vault 测试**
   - 生命周期测试
   - 持久化测试
   - 并发安全测试

4. **配置前端测试环境**
   - Vitest + jsdom
   - React Testing Library
   - Tauri IPC mock

5. **编写组件测试**
   - Button, Input, Card
   - GlassCard, AppBar
   - Dialog, Toast

6. **编写 Store 测试**
   - authStore
   - vaultStore
   - profileStore

7. **配置 E2E 测试**
   - Playwright
   - 多浏览器测试
   - 截图/录屏

8. **配置 CI/CD**
   - GitHub Actions
   - 覆盖率报告
   - 自动测试
