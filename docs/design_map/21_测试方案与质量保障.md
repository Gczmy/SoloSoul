### 2.2 Vault 测试（使用 tempfile）

```rust
#[tokio::test]
async fn test_vault_lifecycle() {
    let temp = tempdir().unwrap();
    let vault = VaultManager::new(temp.path()).unwrap();
    vault.initialize("test", "password").await.unwrap();
    assert_eq!(vault.state().await, VaultState::Unlocked);
    vault.lock().await.unwrap();
    assert_eq!(vault.state().await, VaultState::Locked);
    vault.unlock("test", "password").await.unwrap();
    assert_eq!(vault.state().await, VaultState::Unlocked);
}

#[tokio::test]
async fn test_vault_wrong_password() {
    let result = vault.unlock("test", "wrong").await;
    assert!(result.is_err());
}
```

## 6. CI/CD（GitHub Actions）

```yaml
jobs:
  rust-check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - run: cargo fmt --check
      - run: cargo clippy --workspace -- -D warnings
      - run: cargo test --workspace --verbose

  frontend-check:
    runs-on: ubuntu-latest
    steps:
      - run: npm ci
      - run: npx tsc --noEmit
      - run: npm run lint
      - run: npx vitest run --coverage

  e2e-test:
    runs-on: macos-latest
    steps:
      - run: npm run tauri build -- --debug
      - run: npx playwright test
```
